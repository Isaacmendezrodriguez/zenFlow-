        // Simple JS to handle the widget state toggle between Active Timer and Stop Confirmation
        function toggleWidgetState() {
            const activeState = document.getElementById('widget-active-state');
            const confirmState = document.getElementById('widget-confirm-state');
            
            if (activeState.classList.contains('hidden')) {
                // Show Active, Hide Confirm
                confirmState.classList.add('hidden');
                confirmState.classList.remove('flex');
                
                activeState.classList.remove('hidden');
                activeState.classList.add('flex');
            } else {
                // Hide Active, Show Confirm
                activeState.classList.add('hidden');
                activeState.classList.remove('flex');
                
                confirmState.classList.remove('hidden');
                confirmState.classList.add('flex');
            }
        }
        
        // Optional: Simple logic to make the colon blink
        setInterval(() => {
            const colon = document.querySelector('.animate-pulse');
            if(colon) {
                colon.style.opacity = colon.style.opacity === '0' ? '0.5' : '0';
            }
        }, 1000);
    </script>
