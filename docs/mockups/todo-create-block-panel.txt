<script>
        // Micro-interaction: Auto-calculate duration simulation
        const startInput = document.querySelector('input[type="time"]:nth-of-type(1)');
        const endInput = document.querySelector('input[type="time"]:nth-of-type(2)');
        const durationDisplay = document.querySelector('.font-bold.text-primary');

        function updateDuration() {
            if (startInput.value && endInput.value) {
                const start = new Date(`1970-01-01T${startInput.value}:00`);
                const end = new Date(`1970-01-01T${endInput.value}:00`);
                let diff = (end - start) / 1000 / 60 / 60;
                if (diff < 0) diff += 24;
                durationDisplay.textContent = diff.toFixed(1).replace('.0', '') + 'h';
            }
        }

        [startInput, endInput].forEach(el => el.addEventListener('change', updateDuration));
    </script>
