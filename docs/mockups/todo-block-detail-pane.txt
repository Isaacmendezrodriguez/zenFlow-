        // Micro-interactions
        document.addEventListener('DOMContentLoaded', () => {
            const panel = document.getElementById('detail-panel');
            
            // Simulating entrance animation
            panel.style.transform = 'translateX(100%)';
            setTimeout(() => {
                panel.style.transform = 'translateX(0)';
            }, 100);

            // Close logic placeholder
            const closeBtn = panel.querySelector('[data-icon="close"]').parentElement;
            closeBtn.onclick = () => {
                panel.style.transform = 'translateX(100%)';
                setTimeout(() => {
                    alert('Navegando de vuelta al flujo principal...');
                }, 300);
            };

            // Interactions for buttons
            const buttons = document.querySelectorAll('button');
            buttons.forEach(btn => {
                btn.addEventListener('mouseenter', () => {
                    // Slight lift effect on hover as per style guide
                    if (!btn.classList.contains('bg-primary')) {
                        btn.style.boxShadow = '0 4px 12px rgba(0,0,0,0.05)';
                    }
                });
                btn.addEventListener('mouseleave', () => {
                    btn.style.boxShadow = 'none';
                });
            });
        });
