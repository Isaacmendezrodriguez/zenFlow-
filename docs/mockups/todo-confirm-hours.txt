<script>
    // Simple state management for the modal demo
    const hoursInput = document.getElementById('hours-input');
    const displayReal = document.getElementById('display-real');
    const impactText = document.querySelector('.bg-primary-fixed .font-bold');

    hoursInput.addEventListener('input', (e) => {
      const val = e.target.value || 0;
      displayReal.innerText = val;
      impactText.innerText = val + 'h';
    });

    // Handle ESC key to close modal
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        document.getElementById('confirmation-modal').classList.add('hidden');
      }
    });
  </script>
